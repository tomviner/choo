# Choo CLI Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a CLI called `choo` to the existing traintimes SDK, providing `next`, `board`, `service`, `alias`, and `auth` commands with Rich-formatted output.

**Architecture:** The CLI is a thin Typer layer over the existing SDK. New code lives in `choo/` (separate package within the same repo). Config management, station resolution, and output formatting are isolated modules. The existing `traintimes/` SDK gets a small refactor to support lazy auth loading.

**Tech Stack:** Typer, Rich, rapidfuzz, platformdirs (XDG), existing traintimes SDK (Pydantic, requests)

**Spec:** `docs/superpowers/specs/2026-03-14-cli-design.md`

**API notes (from RTT docs):**
- Station identifiers can be CRS or TIPLOC (not just CRS)
- `countdownMinutes` only appears on high-frequency metro services — cannot rely on it universally; compute countdown from departure time when absent
- `plannedCancel` services should be disregarded — filter out by default in next/board views
- Negative lateness values mean early arrivals — display as "1 min early" not "-1 min"
- API omits default values and empty fields to save bandwidth — all model fields need sensible defaults

```python
"""Configuration management using XDG base directories."""

import json
import os
from pathlib import Path

from platformdirs import user_config_dir


RESERVED_NAMES = {"next", "board", "service", "alias", "auth"}

_APP_NAME = "choo"


def _config_dir() -> Path:
    return Path(user_config_dir(_APP_NAME))


def get_auth() -> tuple[str, str] | None:
    """Get API credentials. Priority: CHOO_AUTH > RTT_AUTH > config file."""
    auth_str = os.environ.get("CHOO_AUTH") or os.environ.get("RTT_AUTH")
    if auth_str:
        return tuple(auth_str.split(":"))

    auth_file = _config_dir() / "auth"
    if auth_file.exists():
        return tuple(auth_file.read_text().strip().split(":"))

    return None


def get_aliases() -> dict[str, str]:
    """Get all aliases. Env vars (CHOO_ALIAS_*) override config file."""
    aliases = {}

    # Load from config file first
    alias_file = _config_dir() / "aliases.json"
    if alias_file.exists():
        aliases = json.loads(alias_file.read_text())

    # Overlay env vars (higher priority)
    prefix = "CHOO_ALIAS_"
    for key, value in os.environ.items():
        if key.startswith(prefix):
            name = key[len(prefix):].lower()
            aliases[name] = value

    return aliases


def set_alias(name: str, crs: str) -> None:
    """Save an alias to the config file."""
    if name.lower() in RESERVED_NAMES:
        raise ValueError(
            f"'{name}' is a reserved command name. "
            f"Choose a different alias name."
        )
    config_dir = _config_dir()
    config_dir.mkdir(parents=True, exist_ok=True)
    alias_file = config_dir / "aliases.json"

    aliases = {}
    if alias_file.exists():
        aliases = json.loads(alias_file.read_text())

    aliases[name.lower()] = crs.upper()
    alias_file.write_text(json.dumps(aliases, indent=2) + "\n")


def remove_alias(name: str) -> None:
    """Remove an alias from the config file."""
    alias_file = _config_dir() / "aliases.json"
    if not alias_file.exists():
        return
    aliases = json.loads(alias_file.read_text())
    aliases.pop(name.lower(), None)
    alias_file.write_text(json.dumps(aliases, indent=2) + "\n")
```

```python
"""Bundled station list with fuzzy matching."""

import csv
from importlib import resources
from functools import lru_cache

from rapidfuzz import fuzz, process


@lru_cache
def load_stations() -> dict[str, str]:
    """Load CRS → name mapping from bundled CSV."""
    stations = {}
    data_file = resources.files("choo._data").joinpath("stations.csv")
    with resources.as_file(data_file) as path:
        with open(path) as f:
            reader = csv.DictReader(f)
            for row in reader:
                stations[row["CRS"]] = row["Name"]
    return stations


def fuzzy_search(query: str, limit: int = 5) -> list[tuple[str, str, float]]:
    """Return [(name, crs, score), ...] sorted by match quality."""
    stations = load_stations()
    name_to_crs = {name: crs for crs, name in stations.items()}
    results = process.extract(
        query, name_to_crs.keys(), scorer=fuzz.WRatio, limit=limit,
    )
    return [(name, name_to_crs[name], score) for name, score, _ in results]
```

```python
"""Station name resolution: alias → CRS → fuzzy match."""

import re

from choo.config import get_aliases
from choo.stations import fuzzy_search, load_stations


class StationNotFound(Exception):
    def __init__(self, query: str, suggestions: list[tuple[str, str, float]]):
        self.query = query
        self.suggestions = suggestions
        super().__init__(f"Station not found: {query!r}")


class AmbiguousStation(Exception):
    def __init__(self, query: str, matches: list[tuple[str, str, float]]):
        self.query = query
        self.matches = matches
        super().__init__(f"Multiple matches for {query!r}")


_CRS_RE = re.compile(r"^[A-Z]{3}$")
_FUZZY_THRESHOLD = 80
_AMBIGUITY_GAP = 10


def resolve_station(input_: str) -> str:
    """Resolve user input to a CRS code.

    Resolution order:
    1. Env aliases (CHOO_ALIAS_*)
    2. Config file aliases
    3. 3 uppercase letters → CRS code
    4. Fuzzy match against station names
    """
    # 1 & 2: Check aliases
    aliases = get_aliases()
    if input_.lower() in aliases:
        return aliases[input_.lower()]

    # 3: CRS code passthrough
    if _CRS_RE.match(input_):
        return input_

    # 4: Fuzzy match
    results = fuzzy_search(input_, limit=5)
    if not results:
        raise StationNotFound(input_, [])

    best_name, best_crs, best_score = results[0]

    if best_score < _FUZZY_THRESHOLD:
        raise StationNotFound(input_, results)

    # Check for ambiguity: if top 2 scores are close, it's ambiguous
    if len(results) > 1:
        _, _, second_score = results[1]
        if best_score - second_score < _AMBIGUITY_GAP:
            close_matches = [
                (n, c, s) for n, c, s in results if best_score - s < _AMBIGUITY_GAP
            ]
            raise AmbiguousStation(input_, close_matches)

    return best_crs
```

```python
"""Rich-formatted output for CLI commands."""

import sys

from rich.console import Console
from rich.table import Table
from rich.text import Text

from traintimes.models import (
    LocationEvent,
    LocationResponse,
    LocationService,
    ServiceLocationState,
    ServiceResponse,
)


def stderr_console() -> Console:
    """Console that writes to stderr for query interpretation."""
    return Console(stderr=True, style="dim")


def _status_text(lateness: int | None, cancelled: bool = False) -> Text:
    """Format delay/status as colored text."""
    if cancelled:
        return Text("Cancelled", style="bold red")
    if lateness is None:
        return Text("", style="dim")
    if lateness < 0:
        return Text(f"{-lateness} min early", style="green")
    if lateness == 0:
        return Text("On time", style="green")
    if lateness <= 5:
        return Text(f"+{lateness} min", style="yellow")
    return Text(f"+{lateness} min", style="bold red")


def _departure_time(event: LocationEvent) -> str:
    """Extract display departure time from a location event."""
    time_str = (
        event.realtime_departure
        or event.gbtt_booked_departure
        or event.realtime_arrival
        or event.gbtt_booked_arrival
        or ""
    )
    if len(time_str) == 4:
        return f"{time_str[:2]}:{time_str[2:]}"
    return time_str


def _dest_name(service: LocationService) -> str:
    if service.destination:
        return service.destination[0].description
    return ""


def format_next(
    console: Console,
    response: LocationResponse,
    to_station: str | None = None,
) -> None:
    """Compact 'next train' display for 1-3 services."""
    # Filter out planned cancellations
    services = [s for s in response.services if not s.planned_cancel]
    if not services:
        console.print("No trains found.")
        return

    markers = ["❶", "❷", "❸", "❹", "❺"]
    for i, svc in enumerate(services):
        event = svc.location_detail
        time = _departure_time(event)
        plat = f"Plat {event.platform}" if event.platform else ""
        lateness = event.realtime_gbtt_departure_lateness
        is_cancelled = svc.planned_cancel or (
            event.display_as and "CANCELLED" in event.display_as.value
        )
        status = _status_text(lateness, cancelled=bool(is_cancelled))
        countdown = f"({svc.countdown_minutes} min)" if svc.countdown_minutes is not None else ""
        marker = markers[i] if i < len(markers) else f"[{i+1}]"

        line = Text()
        line.append(f"  {marker} ", style="bold magenta")
        line.append(f"{time}  ", style="bold")
        line.append(f"{plat:<8}", style="cyan")
        line.append_text(status)
        line.append(f"   {countdown}", style="dim")
        console.print(line)


def format_board(console: Console, response: LocationResponse) -> None:
    """Departure/arrival board table."""
    table = Table(show_header=True, header_style="bold", box=None, pad_edge=False)
    table.add_column("TIME", style="bold", width=7)
    table.add_column("DEST", min_width=14)
    table.add_column("PLAT", width=5, justify="center")
    table.add_column("STATUS", min_width=10)
    table.add_column("OPERATOR", style="dim", min_width=10)
    table.add_column("UID", style="dim cyan")

    for svc in response.services:
        event = svc.location_detail
        time = _departure_time(event)
        dest = _dest_name(svc)
        plat = event.platform or ""
        lateness = event.realtime_gbtt_departure_lateness
        is_cancelled = svc.planned_cancel or (
            event.display_as and "CANCELLED" in event.display_as.value
        )
        status = _status_text(lateness, cancelled=bool(is_cancelled))

        table.add_row(time, dest, plat, status, svc.atoc_name, svc.service_uid)

    if not response.services:
        console.print("No trains found.")
        return

    console.print(table)


def format_service(console: Console, response: ServiceResponse) -> None:
    """Full calling pattern for a single service."""
    origin = response.origin[0].description if response.origin else "?"
    dest = response.destination[0].description if response.destination else "?"
    console.print(f"  {origin} → {dest}\n", style="dim")

    for loc in response.locations:
        name = (loc.description or loc.tiploc)[:20].ljust(20)

        # Determine arr/dep/pass
        if loc.display_as and loc.display_as.value in ("ORIGIN", "STARTS"):
            kind = "dep"
        elif loc.display_as and loc.display_as.value in ("DESTINATION", "TERMINATES"):
            kind = "arr"
        elif loc.gbtt_booked_departure:
            kind = "dep"
        elif loc.gbtt_booked_arrival:
            kind = "arr"
        else:
            kind = "pas"

        booked = loc.gbtt_booked_departure or loc.gbtt_booked_arrival or loc.wtt_booked_pass or ""
        if len(booked) == 4:
            booked = f"{booked[:2]}:{booked[2:]}"

        actual = loc.realtime_departure or loc.realtime_arrival or loc.realtime_pass or ""
        if len(actual) == 4:
            actual = f"{actual[:2]}:{actual[2:]}"

        # Progress indicator
        indicator = ""
        if loc.service_location:
            state_labels = {
                ServiceLocationState.AT_PLATFORM: "◀ AT PLATFORM",
                ServiceLocationState.APPROACHING_PLATFORM: "◀ APPROACHING",
                ServiceLocationState.APPROACHING_STATION: "◀ APPROACHING",
                ServiceLocationState.DEPARTURE_PREPARING: "◀ DEP PREP",
                ServiceLocationState.DEPARTURE_READY: "◀ DEP READY",
            }
            indicator = state_labels.get(loc.service_location, "")
        elif actual and actual == booked:
            indicator = "✓"

        line = Text()
        line.append(f"  {name} ", style="")
        line.append(f"{kind} ", style="dim")
        line.append(f"{booked:>5}  ", style="")

        if actual:
            lateness = (
                loc.realtime_gbtt_departure_lateness
                or loc.realtime_gbtt_arrival_lateness
                or 0
            )
            style = "green" if lateness == 0 else ("yellow" if lateness <= 5 else "bold red")
            line.append(f"{actual:>5}", style=style)
        else:
            line.append("     ", style="")

        if indicator:
            style = "bold cyan" if "◀" in indicator else "green"
            line.append(f"  {indicator}", style=style)

        console.print(line)
```

```python
def _resolve_or_exit(station: str) -> str:

def _build_when(
    at: str | None, on: str | None,
) -> datetime.date | datetime.datetime | None:
    """Combine --at and --on flags into a when value for the SDK."""
    date = None
    time = None

    if on:
        on_lower = on.lower()
        if on_lower == "today":
            date = datetime.date.today()
        elif on_lower == "tomorrow":
            date = datetime.date.today() + datetime.timedelta(days=1)
        else:
            # Try day name
            days = {
                "monday": 0, "tuesday": 1, "wednesday": 2, "thursday": 3,
                "friday": 4, "saturday": 5, "sunday": 6,
            }
            if on_lower in days:
                today = datetime.date.today()
                target = days[on_lower]
                delta = (target - today.weekday()) % 7
                if delta == 0:
                    delta = 7  # Next week if same day
                date = today + datetime.timedelta(days=delta)
            else:
                date = datetime.date.fromisoformat(on)

    if at:
        parts = at.split(":")
        time = datetime.time(int(parts[0]), int(parts[1]))

    if date and time:
        return datetime.datetime.combine(date, time)
    if date:
        return date
    if time:
        return datetime.datetime.combine(datetime.date.today(), time)
    return None


def _print_interpretation(

    to_name: str | None = None,

    elif isinstance(when, datetime.date):
        parts.append(f", {when:%a %-d %b}")
    else:
        parts.append(", now")

    err.print("".join(parts))
```
